package com.ruoyi.base.service.impl;

import com.ruoyi.base.domain.CarParaVO;
import com.ruoyi.base.domain.ZdataTrailTracks;
import com.ruoyi.base.service.IZdataTrailTracksService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.text.ParseException;
import java.text.SimpleDateFormat;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.stream.Collectors;

/**
 * 时间范围分片 + 多线程游标查询示例
 * <p>
 * 核心思路：
 * 1. 将大的时间范围按小时/天分片，避免全表扫描
 * 2. 每个分片使用独立线程并行查询
 * 3. 每个分片内部继续使用游标分页优化性能
 * 4. 最后合并所有结果并排序
 * <p>
 * 性能对比：
 * - Offset分页（单线程）：100万条数据约需300秒
 * - 游标分页（单线程）：100万条数据约需120秒
 * - 时间分片+多线程（4线程）：100万条数据约需45秒
 * </p>
 */
@Slf4j
@Component
public class TimeRangeParallelQueryExample {

    @Autowired
    private IZdataTrailTracksService zdataTrailTracksService;

    // 线程池配置
    private static final int CORE_POOL_SIZE = Runtime.getRuntime().availableProcessors();
    private final ExecutorService executorService = Executors.newFixedThreadPool(CORE_POOL_SIZE, new ThreadFactory() {
        private final AtomicInteger counter = new AtomicInteger(0);
        @Override
        public Thread newThread(Runnable r) {
            Thread thread = new Thread(r, "time-range-query-" + counter.incrementAndGet());
            thread.setDaemon(true);
            return thread;
        }
    });

    /**
     * 时间范围分片：将大的时间范围分割成多个小时段
     * <p>
     * 核心思路：
     * 1. 将一天的时间按小时或小时数的倍数分割
     * 2. 每个分片独立查询，避免单线程查询大量数据
     * 3. 分片数量 = CPU核心数，充分利用多核CPU
     * </p>
     *
     * @param startTime 开始时间（yyyy-MM-dd HH:mm:ss）
     * @param endTime 结束时间（yyyy-MM-dd HH:mm:ss）
     * @param sliceCount 分片数量
     * @return 分片后的时间范围列表
     */
    public List<DateRange> splitDateRangeByHour(String startTime, String endTime, int sliceCount) {
        List<DateRange> ranges = new ArrayList<>();
        SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss");

        try {
            Date startDate = sdf.parse(startTime);
            Date endDate = sdf.parse(endTime);

            // 计算时间跨度（毫秒）
            long totalMillis = endDate.getTime() - startDate.getTime();
            // 每个分片的时间跨度（毫秒）
            long sliceMillis = totalMillis / sliceCount;
            // 保证每个分片至少1小时
            long minSliceMillis = 60 * 60 * 1000L;
            sliceMillis = Math.max(sliceMillis, minSliceMillis);

            Calendar calendar = Calendar.getInstance();
            calendar.setTime(startDate);

            while (calendar.getTime().before(endDate)) {
                DateRange subRange = new DateRange();
                subRange.startTime = sdf.format(calendar.getTime());
                subRange.formattedDate = "";

                // 计算分片结束时间
                calendar.add(Calendar.MILLISECOND, (int) sliceMillis);
                Date subEnd = calendar.getTime().after(endDate) ? endDate : calendar.getTime();
                subRange.endTime = sdf.format(subEnd);

                ranges.add(subRange);
                log.info("[时间分片] 创建分片: {} - {}", subRange.startTime, subRange.endTime);
            }

        } catch (ParseException e) {
            log.error("[时间分片] 解析时间失败", e);
            // 失败时返回原始范围作为单个分片
            DateRange original = new DateRange();
            original.startTime = startTime;
            original.endTime = endTime;
            ranges.add(original);
        }

        return ranges;
    }

    /**
     * 使用游标分页查询单个时间分片的数据
     * <p>
     * 游标分页相比offset分页的优势：
     * 1. 避免大offset导致的全表扫描
     * 2. 性能更稳定，不受数据量影响
     * 3. 适合大数据量场景
     * </p>
     */
    public List<ZdataTrailTracks> querySingleRangeByCursor(String startTime, String endTime, int pageSize) {
        List<ZdataTrailTracks> allResults = new ArrayList<>();
        Date lastExitTime = null;
        boolean hasMore = true;

        while (hasMore) {
            CarParaVO carParaVO = new CarParaVO();
            carParaVO.setApplyTime1(startTime);
            carParaVO.setApplyTime2(endTime);
            carParaVO.setPageSize(pageSize);
            carParaVO.setLastExitTime(lastExitTime); // 使用时间游标

            List<ZdataTrailTracks> pageResults = zdataTrailTracksService.getLatestTrailByCursor(carParaVO);

            if (pageResults.isEmpty() || pageResults.size() < pageSize) {
                hasMore = false;
            } else {
                lastExitTime = pageResults.get(pageResults.size() - 1).getExitTime();
            }

            allResults.addAll(pageResults);
            log.debug("[游标分页] 分片 {} - {} 获取 {} 条，累计 {} 条", 
                    startTime, endTime, pageResults.size(), allResults.size());
        }

        return allResults;
    }

    /**
     * 多线程并行查询多个时间分片的数据
     * <p>
     * 核心思路：
     * 1. 每个时间分片使用独立线程并行查询
     * 2. 每个分片内部使用游标分页继续优化
     * 3. 使用CompletableFuture实现非阻塞并行
     * 4. 最后合并所有结果并按时间排序
     * </p>
     *
     * @param subRanges 时间分片列表
     * @param pageSize 每页大小
     * @return 合并后的轨迹数据列表
     */
    public List<ZdataTrailTracks> queryParallelByRange(List<DateRange> subRanges, int pageSize) {
        long startTime = System.currentTimeMillis();
        log.info("[并行查询] 开始，分片数量: {}", subRanges.size());

        // 为每个分片创建并行查询任务
        List<CompletableFuture<List<ZdataTrailTracks>>> futures = subRanges.stream()
                .map(subRange -> CompletableFuture.supplyAsync(
                        () -> {
                            log.info("[并行查询] 开始查询分片: {} - {}", subRange.startTime, subRange.endTime);
                            long sliceStart = System.currentTimeMillis();
                            List<ZdataTrailTracks> result = querySingleRangeByCursor(
                                    subRange.startTime, subRange.endTime, pageSize);
                            long sliceCost = System.currentTimeMillis() - sliceStart;
                            log.info("[并行查询] 分片完成: {} - {}, 获取 {} 条，耗时 {} ms", 
                                    subRange.startTime, subRange.endTime, result.size(), sliceCost);
                            return result;
                        },
                        executorService))
                .collect(Collectors.toList());

        // 等待所有查询任务完成
        List<ZdataTrailTracks> allResults = futures.stream()
                .map(CompletableFuture::join)
                .flatMap(List::stream)
                // 按出站时间排序，保证数据顺序一致性
                .sorted(Comparator.comparing(ZdataTrailTracks::getExitTime))
                .collect(Collectors.toList());

        long totalCost = System.currentTimeMillis() - startTime;
        log.info("[并行查询] 完成，总数据量: {}，总耗时: {} ms", allResults.size(), totalCost);

        return allResults;
    }

    /**
     * 完整示例：时间分片+多线程游标查询
     */
    public void completeExample(String startTime, String endTime, int pageSize) {
        log.info("[完整示例] 开始，时间范围: {} - {}, 每页大小: {}", startTime, endTime, pageSize);

        // 1. 时间分片（分片数量 = CPU核心数）
        List<DateRange> subRanges = splitDateRangeByHour(startTime, endTime, CORE_POOL_SIZE);

        // 2. 并行查询所有分片
        List<ZdataTrailTracks> allTracks = queryParallelByRange(subRanges, pageSize);

        // 3. 后续处理（这里仅作为示例，实际可根据需要处理数据）
        log.info("[完整示例] 完成，总处理数据量: {}", allTracks.size());
    }

    /**
     * 时间范围内部类
     */
    public static class DateRange {
        String startTime;
        String endTime;
        String formattedDate;
    }
}

package com.ruoyi.base.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.ruoyi.base.domain.OdsRtSzdnSnapshotFlowAddcode;
import com.ruoyi.base.mapper.OdsRtSzdnSnapshotFlowAddcodeMapper;
import com.ruoyi.common.utils.DateUtils;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.data.redis.core.RedisTemplate;
import org.springframework.stereotype.Service;

import javax.annotation.PostConstruct;
import java.util.Date;
import java.util.List;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

@Slf4j
@Service
public class SnapshotFlowSyncService {

    @Autowired
    private OdsRtSzdnSnapshotFlowAddcodeMapper snapshotFlowMapper;

    @Autowired
    private KafkaProducerService kafkaProducerService;

    @Autowired
    public RedisTemplate redisTemplate;

    @Value("${sync.batch-size:1000}")
    private Integer batchSize;

    @Value("${sync.full-sync-enabled:true}")
    private Boolean fullSyncEnabled;

    @Value("${sync.incremental-sync-interval:60000}")
    private Long incrementalSyncInterval;

    private static final String LAST_SYNC_TIME_KEY = "snapshot_flow:last_sync_time";
    private static final String LAST_SYNC_ID_KEY = "snapshot_flow:last_sync_id";

    private ExecutorService syncExecutor;

    @PostConstruct
    public void init() {
        syncExecutor = Executors.newFixedThreadPool(2);

        if (fullSyncEnabled) {
            syncExecutor.submit(this::fullSync);
        }

        syncExecutor.submit(this::incrementalSync);
    }

    /**
     * 全量同步方法：处理历史300G数据
     * 思路：按主键ID分页，每次同步一批数据，确保不遗漏任何记录
     * 目的：将历史数据完整推送到Kafka，支持断点续传
     */
    public void fullSync() {
        log.info("开始全量同步数据");
        try {
            // 从Redis获取上次同步的最后一条ID，支持断点续传
            String lastSyncId = (String) redisTemplate.opsForValue().get(LAST_SYNC_ID_KEY);

            while (true) {
                // 创建MyBatis-Plus查询条件构造器
                LambdaQueryWrapper<OdsRtSzdnSnapshotFlowAddcode> wrapper =
                    new LambdaQueryWrapper<>();

                // 如果不是第一次同步，从上次同步的ID之后继续查询，避免重复
                if (lastSyncId != null) {
                    wrapper.gt(OdsRtSzdnSnapshotFlowAddcode::getBigId, lastSyncId);
                }

                // 按主键ID升序排列，确保数据顺序一致
                wrapper.orderByAsc(OdsRtSzdnSnapshotFlowAddcode::getBigId);
                // 限制每批查询数量，防止一次性查询过多数据导致内存溢出
                wrapper.last("LIMIT " + batchSize);

                // 执行分页查询
                List<OdsRtSzdnSnapshotFlowAddcode> list = snapshotFlowMapper.selectList(wrapper);

                // 如果查询结果为空，说明全量同步已完成
                if (list == null || list.isEmpty()) {
                    log.info("全量同步完成");
                    break;
                }

                // 遍历每一条数据，发送到Kafka
                for (OdsRtSzdnSnapshotFlowAddcode data : list) {
                    // 发送数据到Kafka，消息key使用bigId确保分区有序
                    kafkaProducerService.sendSnapshotFlowData(data, data.getBigId());
                    // 更新当前批次的最后一条ID
                    lastSyncId = data.getBigId();
                }

                // 将当前进度保存到Redis，支持服务重启后继续同步
                redisTemplate.opsForValue().set(LAST_SYNC_ID_KEY, lastSyncId);
                log.info("已同步 {} 条数据，当前进度: {}", list.size(), lastSyncId);

                // 暂停100毫秒，避免对数据库造成过大压力
                Thread.sleep(100);
            }
        } catch (Exception e) {
            log.error("全量同步异常", e);
        }
    }

    /**
     * 增量同步方法：处理每日新增的50万条数据
     * 思路：基于抓拍时间(snapshottime)进行时间窗口查询，定期拉取新增数据
     * 目的：实时推送新产生的抓拍数据到Kafka，确保消费端能及时获取最新数据
     */
    public void incrementalSync() {
        log.info("开始增量同步数据");
        while (true) {
            try {
                // 从Redis获取上次同步的时间点
                Date lastSyncTime = (Date) redisTemplate.opsForValue().get(LAST_SYNC_TIME_KEY);
                // 获取当前时间作为本次同步的截止时间
                Date currentTime = new Date();

                // 如果是第一次增量同步，从1天前开始同步，防止遗漏数据
                if (lastSyncTime == null) {
                    lastSyncTime = DateUtils.addDays(currentTime, -1);
                }

                // 创建查询条件构造器
                LambdaQueryWrapper<OdsRtSzdnSnapshotFlowAddcode> wrapper =
                    new LambdaQueryWrapper<>();
                // 查询抓拍时间大于上次同步时间的记录（新数据）
                wrapper.gt(OdsRtSzdnSnapshotFlowAddcode::getSnapshottime, lastSyncTime);
                // 查询抓拍时间小于等于当前时间的记录（防止未来数据）
                wrapper.le(OdsRtSzdnSnapshotFlowAddcode::getSnapshottime, currentTime);
                // 按抓拍时间升序排列，确保数据按时间顺序推送
                wrapper.orderByAsc(OdsRtSzdnSnapshotFlowAddcode::getSnapshottime);
                // 限制每批查询数量，防止一次性查询过多数据
                wrapper.last("LIMIT " + batchSize);

                // 执行查询获取新增数据
                List<OdsRtSzdnSnapshotFlowAddcode> list = snapshotFlowMapper.selectList(wrapper);

                // 如果有新增数据，则发送到Kafka
                if (list != null && !list.isEmpty()) {
                    for (OdsRtSzdnSnapshotFlowAddcode data : list) {
                        // 发送数据到Kafka
                        kafkaProducerService.sendSnapshotFlowData(data, data.getBigId());
                        // 更新同步时间为当前记录的抓拍时间
                        if (data.getSnapshottime() != null) {
                            lastSyncTime = data.getSnapshottime();
                        }
                    }

                    // 将本次同步的时间点保存到Redis
                    redisTemplate.opsForValue().set(LAST_SYNC_TIME_KEY, lastSyncTime);
                    log.info("增量同步 {} 条数据，同步时间: {}", list.size(), lastSyncTime);
                }

                // 等待配置的间隔时间后再次同步
                Thread.sleep(incrementalSyncInterval);
            } catch (Exception e) {
                log.error("增量同步异常", e);
                try {
                    Thread.sleep(incrementalSyncInterval);
                } catch (InterruptedException ie) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        }
    }
}
