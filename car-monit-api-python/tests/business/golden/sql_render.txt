            t.envehicleid plateKey,
            t.enaxlecount,
            t.entime enterTime,
            t.entolllaneid,
            t.entollstationid,
            t.entollstationname,
            t.envehicleclass,
            t.envehicleid,
            t.envehicletype,
            t.enweight,
            t.extime exitTime,
            t.extolllaneid,
            t.extollstationid,
            t.extollstationname,
            t.exvehicleclass,
            t.exvehicleid,
            t.exvehicletype,
            t.exweight,
            t.trailpoints,
            t.trailtype
        from jl_data01.zdata_trail_fitting t WHERE  t.extime >= to_timestamp(:p0,'yyyy-mm-dd HH24:MI:SS')
         and t.extime <= to_timestamp(:p1,'yyyy-mm-dd HH24:MI:SS') order by t.cid,t.extime

-- com.ruoyi.base.mapper.ZdataTrailTracksMapper.getUnFitting
select
        t.platenumber plateKey,
        t.entime enterTime,
        t.extime exitTime,
        t.trailpoints,
        t.vehicletype envehicletype
        from jl_data01.zdata_trail_unfitting t WHERE  t.capturedate >= to_timestamp(:p0,'yyyy-mm-dd HH24:MI:SS')
            and t.capturedate <= to_timestamp(:p1,'yyyy-mm-dd HH24:MI:SS') order by t.capturedate,t.platenumber

        t.type,
        t.point_count,
        t.plate_key,
        t.enter_time,
        t.exit_time,
        t.enter_locationcode,
        t.exit_locationcode,
        t.trail_points,
        t.create_time from jl_data01.zdata_trail_tracks t
         where t.uuid = :p0
